284  switch (c)
285  {
286  case 'n':
287  return '\n';

292  case '\\':
293  return '\\';

360  case '=':

374  case ',':

377  case '"':
378  quote_count++;
379  return String_detect;
380  case '_':
381  appendChar(lexeme, key);
382  return Id_global0;
383  case '!':

385  default:

387  {

389  return Letter;

396  return Start;
397  }
398  else if (key == ' ' || key == '\t' || key == '\r' || key == '\v' || key == '\f')
399  {

401  return Start;
402  }
403  else if (isdigit(key))
404  {

406  if (key == '0')

414  }

427  ungetc(key, stdin);
428  return Start;
429  case Number:
430  appendChar(lexeme, key);
431  if (isdigit(key))
432  {
433  return Number;
434  }
435  else if (key == '.')

437  return Float;
438  }

441  return Exponent;
442  }

444  createToken(INT, lexeme);
445  ungetc(key, stdin);
446  return Start;
447  case Float:
448  appendChar(lexeme, key);
449  if (isdigit(key))
450  {
451  float_detected = true;
452  return Float;
453  }
454  else if ((key == 'e' || key == 'E') && float_detected)
455  {
456  float_detected = false;
457  return Exponent;
458  }else if (!float_detected) // Pokud se za desetinnou tečkou nenachází žádné číslo
459  {

461  ifjexit(ERR_LEX);
462 
463  }

465  createToken(FLOAT, lexeme);
466  ungetc(key, stdin);
467  float_detected = false;
468  return Start;
469 
470  // Exponent
471  case Exponent:
472  appendChar(lexeme, key);
473  if (isdigit(key))
474  {
475  return Exponent_float;
476  }
477  else if (key == '+' || key == '-')
478  {
479  return Exponent_sign;

482  ifjexit(ERR_LEX);
483  return Start;
484  case Exponent_sign:
485  appendChar(lexeme, key);
486  if (isdigit(key))
487  return Exponent_float;

489  fprintf(stderr, "Neplatný desetinný literál na řádku %d\n", g_line);
490  ifjexit(ERR_LEX);
491  return Start;
492  case Exponent_float:
493  appendChar(lexeme, key);
494  if (isdigit(key))
495  {
496  return Exponent_float;
497  }
498  removeLast(lexeme);
499  createToken(FLOAT, lexeme);
500  ungetc(key, stdin);
501  return Start;
502 
503  case Hex:
504  if (isxdigit(key))
505  {
506  appendChar(lexeme, key);
507  return Hex;
508  }
509  if (lexeme->length <= 2) // Pokud je jen "0x"

515  ungetc(key, stdin);
516  return Start;
517 
518  case Not:
519  if (key == '=')
520  {
521  createToken(NOT_EQUAL, lexeme);
522  return Start;
523  }
524  ungetc(key, stdin);
525  return Error;

527  case Letter:

595  }

605  return String_single;
606  }
607  if (key == '\n' || key == EOF)

610  ifjexit(ERR_LEX);
611  }
612  appendChar(lexeme, key);
613  return String_single;
614  case String_multi: // Víceřádkový
615  if (key == EOF)
616  {
617  fprintf(stderr, "Neukončený víceřádkový string na řádku %d\n", g_line);
618  ifjexit(ERR_LEX);
619  }
620  if (key == '"')
621  {
622  quote_count++;
623  return String_q_check2;
624  }
625  quote_count = 0;
626  appendChar(lexeme, key);
627  return String_multi;
628  case String_q_check2: // Kontrola zda uvozovky ukončují víceřádkový řetězec
629  ungetc(key,stdin);
630  if (quote_count == 3)
631  {
632  quote_count = 0;
633  createToken(STRING, lexeme);
634  return Start;
635  }
636  return String_multi;
637 
638  // Built-in funkce
639  case BuiltIn:
640  if (key == ' ' || key == '\t' || key == '\r' || key == '\v' || key == '\f')
641  {
642  // Běžné whitespacy ignorujeme, čekáme na '.' (povolí "Ifj . write")
643  return BuiltIn;
644  }
645  else if (key == '.')
646  {
647  resetBuffer(lexeme);
648  return BuiltIn_dot;
649  }

651  createToken(KEYWORD_Ifj, lexeme);
652  ungetc(key, stdin);
653  return Start;
654  case BuiltIn_dot:
655  if (isspace(key))

658  return BuiltIn_dot;

663  return BuiltIn_fun;

666  {

669  }

689  return Comment_multi;
690  }

692  ungetc(key, stdin);
693  return Start;
694

698  {
699  // Konec řádkového komentáře, uzavři příkaz EOL

701  return Start;
702  }
703  return Comment_single;
704  case Comment_multi: // Víceřádkový
705  if (key == EOF)
706  {
707  fprintf(stderr, "Neukončený víceřádkový komentář na řádku %d\n", g_line);
708  ifjexit(ERR_LEX);
709  }
710  if (key == '*')
711  {
712  return Comment_multi_end; // Možný konec "*/"
713  }
714  if (key == '/')
715  {
716  return Comment_multi_slash; // Možný nový začátek "/*"
717  }
718  if (key == '\n')
719  {
720  // Drž číslování řádků i uvnitř komentářů
721  g_line++;
722  }
723  return Comment_multi;
724  case Comment_multi_slash:
725  if (key == EOF)
726  {
727  fprintf(stderr, "Neukončený víceřádkový komentář na řádku %d\n", g_line);
728  ifjexit(ERR_LEX);
729  }
730  if (key == '*')
731  {
732  // Detekováno "/*" uvnitř komentáře vnoření + 1
733  g_ml_depth++;
734  return Comment_multi;
735  }
736  if (key == '\n')
737  {
738  g_line++;
739  }
740  // Nebylo to "/*", pořád jsme uvnitř
741  return Comment_multi;
742  case Comment_multi_end:
743  if (key == EOF) {
744  fprintf(stderr, "Neukončený víceřádkový komentář (EOF) na řádku %d\n", g_line);
745  ifjexit(ERR_LEX);
746  }
747  if (key == '/')
748  {
749  // Detekováno "*/"
750  g_ml_depth--;
751  return Comment_multi_end_check;
752  }
753  if (key == '*') {
754  return Comment_multi_end; // NE VRÁTIT SE
755  }
756  if (key == '\n')
757  {
758  g_line++;
759  }
760  // Nebylo to "*/", zpět do těla komentáře
761  return Comment_multi;
762  case Comment_multi_end_check:
763  if (g_ml_depth <= 0)
764  {
765  return Start; // Vycházíme z komentáře
766  }
767  return Comment_multi; // Pořád jsme ve vnořeném komentáři
768 
769  case Lesser:
770  if (key == '=')
771  {
772  createToken(LESSER_EQUAL, lexeme);
773  return Start;
774  }
775  createToken(LESSER, lexeme);
776  ungetc(key, stdin);
777  return Start;
778 
779  case Greater:
780  if (key == '=')
781  {
782  createToken(GREATER_EQUAL, lexeme);
783  return Start;
784  }
785  createToken(GREATER, lexeme);
786  ungetc(key, stdin);
787  return Start;
788 
789  case Assign:
790  if (key == '=')
791  {
792  createToken(EQUAL, lexeme);
793  return Start;
794  }
795  createToken(ASSIGN, lexeme);
796  ungetc(key, stdin);
797  return Start;
798 
799  case Error:

801  ifjexit(ERR_LEX);
802  break;

805  return Error;
806  }
