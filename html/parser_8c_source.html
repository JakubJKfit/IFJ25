884 static AstNode *parse_expr_bp(IfjPrec minbp)
885 {
886  int line = ifj_get_line();
887 
888  AstNode *left = parse_primary();
889  if (!left)
890  return NULL;
891 
892  while (1)
893  {
894  int lbp = lbp_of(ifj_cur.type);
895  if (lbp < 0 || lbp < (int)minbp)
896  break;
897 
898  TokenType op = ifj_cur.type;
899  next();
900  IfjPrec rbp = (IfjPrec)(lbp + 1); // left-assoc
901 
902  if (op == KEYWORD_is)
903  {
904  if (!(ifj_cur.type == KEYWORD_Num ||
905  ifj_cur.type == KEYWORD_String ||
906  ifj_cur.type == KEYWORD_Null))
907  {
908  free_ast_node(left);
909  fprintf(stderr, "l%d: right side of 'is' must be type keyword (Num/String/Null)", ifj_get_line());
910  ifjexit(ERR_SYN);
911  }
912 
913  Token type_token = ifj_cur;
914  type_token.lexeme = NULL;
915  next();
916 
917  AstNode *right = create_literal(type_token);
918  left = create_binary_expr(op, left, right, line);
919  continue;
920  }
921 
922  AstNode *right = parse_expr_bp(rbp);
923  if (!right)
924  {
925  free_ast_node(left);
926  return NULL;
927  }
928 
929  left = create_binary_expr(op, left, right, line);
930  }
931  return left;
932 }

986 static void semantic_firstpass(AstNode *node, symstack *stack)
987 {
988 
989  if (node->type != AST_PROGRAM)
990  return;
991 
992  AstNodeProgram *program = (AstNodeProgram *)node;
993  AstNodeList *funcs = program->functions; // seznam vsech funkci
994  tree_node **global_table = stack_top(stack); // globalni tabulka symbolu
995 
996  char identif_buffer[256]; // buffer pro ulozeni identifikatoru s aritou
997  symbol_data func_data;
998  char func_key[256]; // buffer pro ulozeni identifikatoru bez arity
999 
1000  // projdi všechny definice funkcí v programu
1001  for (int i = 0; i < funcs->count; i++)
1002  {
1003  AstNodeFuncDef *func = (AstNodeFuncDef *)funcs->items[i];
1004 
1005  char *func_name = func->func_id.lexeme;
1006  int arity = func->params->count;
1007 
1008  int temp = 0;
1009  switch (func->kind)
1010  {
1011  case FUNC_IS_FUNC:
1012  // klíč pro deklaraci funkce bez arity (pro kontrolu existence)
1013  snprintf(func_key, 256, "%s@F", func_name);
1014 
1015  // klíč pro konkrétní definici s aritou
1016  temp = snprintf(identif_buffer, 256, "%s@%d", func_name, arity);
1017  break;
1018  case FUNC_IS_GETTER:
1019  temp = snprintf(identif_buffer, 256, "%s@GET", func_name);
1020  break;
1021  case FUNC_IS_SETTER:
1022  temp = snprintf(identif_buffer, 256, "%s@SET", func_name);
1023  break;
1024  default:
1025  fprintf(stderr, "Unknown function kind\n");
1026  ifjexit(ERR_INTERNAL);
1027  }
1028  if (temp < 0 || temp >= 256)
1029  {
1030  fprintf(stderr, "Line %d, %s function identifier too long for parsing\n", func->base.line_number, func_name);
1031  ifjexit(ERR_SEM_OTHER);
1032  }
1033 
1034  // kontrola redefince funkce
1035  symbol_data *found;
1036  if (search_symbol(*global_table, identif_buffer, &found))
1037  {
1038  // semanticka chyba redefinice funkce
1039  fprintf(stderr, "Line: %d, redefinition of function %s\n", func->base.line_number, func_name);
1040  ifjexit(ERR_SEM_REDEFINED);
1041  }
1042 
1043  // u běžné funkce si navíc pamatujeme, že pod daným jménem existuje nějaká definice
1044  if (func->kind == FUNC_IS_FUNC)
1045  {
1046  symbol_data *found_decl;
1047  if (!search_symbol(*global_table, func_key, &found_decl))
1048  {
1049  symbol_data data;
1050  data.identifier = func_key;
1051  data.id_type = FUNC_ID;
1052  data.data_type = Undefined;
1053  data.arity = -1;
1054  data.offset = 0;
1055 
1056  insert_symbol(global_table, &data);
1057  }
1058  }
1059 
1060  // ulož záznam pro konkrétní kombinaci jméno+arita
1061  func_data.identifier = identif_buffer;
1062  func_data.id_type = FUNC_ID;
1063  func_data.data_type = Undefined; // návratový typ určí až druhý průchod
1064  func_data.arity = arity;
1065  func_data.offset = 0;
1066 
1067  insert_symbol(global_table, &func_data);
1068  }
1069 }

1086 static void semantic_recurs(AstNode *node, symstack *stack, int *current_offset, symbol_data *current_func_data)
1087 {
1088  // konec analyzy
1089  if (!node)
1090  return;
1091 
1092  // switch podle typu AST uzlu
1093  switch (node->type)
1094  {
1095  case AST_PROGRAM:
1096  {
1097  AstNodeProgram *program = (AstNodeProgram *)node;
1098  for (int i = 0; i < program->functions->count; i++) // prochazeni definic funkci
1099  {
1100  semantic_recurs(program->functions->items[i], stack, NULL, NULL);
1101  }
1102  break;
1103  }
1104  case AST_FUNC_DEF:
1105  {
1106  AstNodeFuncDef *func = (AstNodeFuncDef *)node;
1107 
1108  int func_offset = 0; // posun promennych
1109  char func_key[256]; // buffer pro identifikator funkce
1110  int arity = func->params->count; // arita funkce
1111  switch (func->kind)
1112  {
1113  case FUNC_IS_FUNC:
1114  snprintf(func_key, 256, "%s@%d", func->func_id.lexeme, arity);
1115  break;
1116  case FUNC_IS_GETTER:
1117  snprintf(func_key, 256, "%s@GET", func->func_id.lexeme);
1118  break;
1119  case FUNC_IS_SETTER:
1120  snprintf(func_key, 256, "%s@SET", func->func_id.lexeme);
1121  break;
1122  }
1123 
1124  symbol_data *func_data;
1125 
1126  // spravne se funkce v firstpass ulozila
1127  if (!search_scopes(stack, func_key, &func_data))
1128  {
1129  fprintf(stderr, "Function %s not found in secondpass\n", func_key);
1130  ifjexit(ERR_INTERNAL);
1131  }
1132 
1133  func_data->data_type = Null; // výchozí návratová hodnota
1134  stack_push(stack, func->func_id.lexeme); // func scope
1135  tree_node **param_table = stack_top(stack);
1136 
1137  // kontrola parametru
1138  for (int i = 0; i < func->params->count; i++)
1139  {
1140  AstNodeVariable *param = (AstNodeVariable *)func->params->items[i];
1141  char *param_name = param->token.lexeme;
1142 
1143  // duplikatni parametry v definici fce
1144  symbol_data *found;
1145  if (search_symbol(*param_table, param_name, &found))
1146  {
1147  fprintf(stderr, "Line: %d, duplicate parameter %s in function definition\n", param->base.line_number, param_name);
1148  ifjexit(ERR_SEM_REDEFINED);
1149  }
1150 
1151  func_offset -= 8; // byte pro param
1152 
1153  // data parametru
1154  symbol_data param_data;
1155  param_data.identifier = param_name;
1156  param_data.id_type = VARIABLE_ID;
1157  param_data.data_type = Undefined;
1158  param_data.offset = func_offset;
1159  param_data.arity = 0; // neni fce
1160 
1161  insert_symbol(param_table, &param_data);
1162 
1163  // AST anotace
1164  param->stack_offset = param_data.offset;
1165  param->data_type = param_data.data_type;
1166  }
1167 
1168  AstNodeBlock *body = func->body;
1169  for (int i = 0; i < body->statements->count; i++)
1170  {
1171  semantic_recurs(body->statements->items[i], stack, &func_offset, func_data);
1172  }
1173  stack_pop(stack);
1174  break;
1175  }
1176  case AST_STMT_BLOCK:
1177  {
1178  AstNodeBlock *block = (AstNodeBlock *)node;
1179 
1180  if (current_offset == NULL)
1181  {
1182  fprintf(stderr, "Block in global scope\n");
1183  ifjexit(ERR_INTERNAL);
1184  }
1185 
1186  stack_push(stack, "block"); // vnoreni do bloku
1187  for (int i = 0; i < block->statements->count; i++)
1188  {
1189  semantic_recurs(block->statements->items[i], stack, current_offset, current_func_data);
1190  }
1191  stack_pop(stack);
1192  break;
1193  }
1194  case AST_STMT_VAR_DECL:
1195  {
1196  AstNodeVarDecl *decl = (AstNodeVarDecl *)node;
1197  char *var_name = decl->var_id.lexeme;
1198 
1199  tree_node **current_table = stack_top(stack); // aktualni vnorenost
1200  symbol_data *found;
1201 
1202  // redefinice promenne
1203  if (search_symbol(*current_table, var_name, &found))
1204  {
1205  fprintf(stderr, "Line> %d, redefinition of variable %s\n", decl->base.line_number, var_name);
1206  ifjexit(ERR_SEM_REDEFINED);
1207  }
1208 
1209  if (current_offset == NULL)
1210  {
1211  fprintf(stderr, "VAR declaretion found in global scope\n");
1212  ifjexit(ERR_INTERNAL);
1213  }
1214 
1215  *current_offset -= 8; // offset pro codegen
1216 
1217  // ulozeni promenne
1218  symbol_data var_data;
1219  var_data.identifier = var_name;
1220  var_data.id_type = VARIABLE_ID;
1221  var_data.data_type = Null;
1222  var_data.offset = *current_offset;
1223  var_data.arity = 0;
1224 
1225  insert_symbol(current_table, &var_data);
1226 
1227  // anotace AST
1228  decl->stack_offset = var_data.offset;
1229  decl->data_type = var_data.data_type;
1230 
1231  break;
1232  }
1233  case AST_EXPR_VARIABLE:
1234  {
1235  AstNodeVariable *var = (AstNodeVariable *)node;
1236  char *var_name = var->token.lexeme;
1237 
1238  // globalni promenna
1239  if (var->token.type == IDENTIFIER_GLOBAL)
1240  {
1241  var->stack_offset = -1; // specialni offset pro global
1242  var->data_type = Undefined;
1243  break;
1244  }
1245 
1246  // promenna
1247  symbol_data *found;
1248  if (search_scopes(stack, var_name, &found))
1249  {
1250 
1251  var->stack_offset = found->offset;
1252  var->data_type = found->data_type;
1253  }
1254  else // getter
1255  {
1256  char getter_key[256];
1257  int temp = snprintf(getter_key, 256, "%s@GET", var_name);
1258  if (temp < 0 || temp >= 256)
1259  {
1260  fprintf(stderr, "Identifier too long for parsing\n");
1261  ifjexit(ERR_SEM_OTHER);
1262  }
1263  if (search_scopes(stack, getter_key, &found))
1264  {
1265  if (found->id_type != FUNC_ID)
1266  {
1267  fprintf(stderr, "%s not FUNC_ID and cant be getter\n", getter_key);
1268  ifjexit(ERR_INTERNAL);
1269  }
1270  var->stack_offset = 0;
1271  var->data_type = found->data_type;
1272  }
1273  else // nedefinovana promenna/getter
1274  {
1275  fprintf(stderr, "Line: %d, undefined var %s\n", var->base.line_number, var_name);
1276  ifjexit(ERR_SEM_UNDEFINED);
1277  }
1278  }
1279  break;
1280  }

1282  {
1283  AstNodeAssignStmt *assign = (AstNodeAssignStmt *)node;
1284 
1285  semantic_recurs(assign->rvalue, stack, current_offset, current_func_data); // zavolat s pravym potomkem
1286  symbol_data_type right_type = Undefined;
1287  switch (assign->rvalue->type) // leva hodnota nabyje datovy typ prave hodnoty
1288  {
1289  case AST_EXPR_LITERAL:
1290  right_type = ((AstNodeLiteral *)assign->rvalue)->data_type;
1291  break;
1292  case AST_EXPR_VARIABLE:
1293  right_type = ((AstNodeVariable *)assign->rvalue)->data_type;
1294  break;
1295  case AST_EXPR_BINARY:
1296  right_type = ((AstNodeBinaryExpr *)assign->rvalue)->data_type;
1297  break;
1298  case AST_FUNC_CALL:
1299  right_type = ((AstNodeFuncCall *)assign->rvalue)->data_type;
1300  break;
1301  default:
1302  fprintf(stderr, "Internal Error: Invalid rvalue in assignment\n");
1303  ifjexit(ERR_INTERNAL);
1304  }
1305 
1306  // analyza leveho potomka
1307  AstNodeVariable *left = assign->lvalue;
1308  char *left_name = left->token.lexeme;
1309  symbol_data *found;
1310 
1311  // globalni promenna
1312  if (left->token.type == IDENTIFIER_GLOBAL)
1313  {
1314  left->stack_offset = -1;
1315  left->data_type = right_type;
1316  break;
1317  }
1318 
1319  // lokalni promenna
1320  if (search_scopes(stack, left_name, &found))
1321  {
1322  if (found->id_type != VARIABLE_ID)
1323  {
1324  fprintf(stderr, "%s can't be non var\n", left_name);
1325  ifjexit(ERR_SEM_OTHER);
1326  }
1327 
1328  left->stack_offset = found->offset;
1329  left->data_type = right_type;
1330  found->data_type = right_type;
1331  }
1332  else // setter
1333  {
1334  char setter_key[256];
1335  int temp = snprintf(setter_key, 256, "%s@SET", left_name);
1336  if (temp < 0 || temp >= 256)
1337  {
1338  fprintf(stderr, "Identifier too long for parsing\n");
1339  ifjexit(ERR_SEM_OTHER);
1340  }
1341  if (search_scopes(stack, setter_key, &found))
1342  {
1343  if (found->id_type != FUNC_ID)
1344  {
1345  fprintf(stderr, "%s not FUNC_ID and cant be setter\n", setter_key);
1346  ifjexit(ERR_INTERNAL);
1347  }
1348  left->stack_offset = 0;
1349  left->data_type = Undefined; // pro setter je to jedno
1350  }
1351  else
1352  {
1353  fprintf(stderr, "Line %d, Cannot assign to undefined var or setter %s\n", left->base.line_number, left_name);
1354  ifjexit(ERR_SEM_UNDEFINED);

1356  }

1358  }

1362  semantic_recurs(if_stmt->condition, stack, current_offset, current_func_data);

1376  AstNodeReturnStmt *ret = (AstNodeReturnStmt *)node;

1385  semantic_recurs(ret->expr, stack, current_offset, current_func_data);

1387  }

1391  }
1392  else if (current_func_data->data_type != return_type && return_type != Undefined &&
1393  current_func_data->data_type != Undefined)

1395  current_func_data->data_type = Undefined; // vraci odlisny typ, tak se nastavi na undefined a pri behu se na to podiva
1396  }
1397 
1398  break;
1399  }
1400  case AST_EXPR_BINARY:
1401  {

1405

1407  symbol_data_type right_type = get_type(binary_expr->right);
1408

1411  {
1412  binary_expr->data_type = Undefined;
1413  break;
1414  }

1418  {
1419  if (left_type == Null || right_type == Null)

1423  binary_expr->base.line_number);
1424  ifjexit(ERR_SEM_INCOMP);
1425  }
1426  }

1428  {

1435  {
1436  binary_expr->data_type = String;
1437  }

1442  }
1443  break;
1444  case MINUS:
1445  case DIVIDE:
1446  if (left_type == Num && right_type == Num)
1447  {
1448  binary_expr->data_type = Num;
1449  }
1450  else
1451  {
1452  fprintf(stderr, "Line %d, Operands for subtraction or division must be both Num\n", binary_expr->base.line_number);
1453  ifjexit(ERR_SEM_INCOMP);
1454  }
1455  break;
1456  case TIMES:
1457  if (left_type == Num && right_type == Num)
1458  {
1459  binary_expr->data_type = Num;
1460  }
1461  else if (left_type == String && right_type == Num)
1462  {
1463  if (binary_expr->right->type == AST_EXPR_LITERAL && ((AstNodeLiteral *)binary_expr->right)->token.type != INT)
1464  {
1465  fprintf(stderr, "Line %d, Operands for string iteration must be an Integer, not Float\n", binary_expr->base.line_number);
1466  ifjexit(ERR_SEM_INCOMP);
1467  }
1468  binary_expr->data_type = String;
1469  }
1470  else
1471  {
1472  fprintf(stderr, "Line %d, Operands for multipication must be both Num or String * Num\n", binary_expr->base.line_number);
1473  ifjexit(ERR_SEM_INCOMP);
1474  }
1475  break;
1476  case LESSER:
1477  case GREATER:
1478  case LESSER_EQUAL:
1479  case GREATER_EQUAL:
1480  if (left_type == Num && right_type == Num)
1481  {
1482  binary_expr->data_type = Num;
1483  }
1484  else
1485  {
1486  fprintf(stderr, "Line %d: Operands for comparison need to be both Num\n", binary_expr->base.line_number);
1487  ifjexit(ERR_SEM_INCOMP);
1488  }
1489  break;
1490  case EQUAL:
1491  case NOT_EQUAL:
1492  binary_expr->data_type = Num;
1493  break;
1494  case KEYWORD_is:
1495  binary_expr->data_type = Num;
1496  break;
1497  default: // asi nenastane
1498  binary_expr->data_type = Undefined;
1499  break;
1500  }
1501 
1502  break;
1503  }
1504  case AST_FUNC_CALL:
1505  {
